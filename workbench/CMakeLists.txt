cmake_minimum_required(VERSION 3.16)
project(uFlipperCore LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(PkgConfig REQUIRED)
pkg_check_modules(SODIUM REQUIRED IMPORTED_TARGET libsodium)
find_package(SQLite3 REQUIRED)
if(TARGET SQLite3::SQLite3)
 set(SQLITE_TARGET SQLite3::SQLite3)
else()
 set(SQLITE_TARGET SQLite::SQLite3)
endif()
add_library(uflipper_core core/operation_coordinator.cpp core/capture_store.cpp core/file_crypto.cpp)
target_include_directories(uflipper_core PUBLIC core)
target_link_libraries(uflipper_core PUBLIC PkgConfig::SODIUM ${SQLITE_TARGET})
target_compile_options(uflipper_core PRIVATE -Wall -Wextra -Wpedantic)
include(CTest)
if(BUILD_TESTING)
 add_executable(uflipper_core_tests ../tests/core/core_tests.cpp)
 target_link_libraries(uflipper_core_tests PRIVATE uflipper_core)
 add_test(NAME core_contract COMMAND uflipper_core_tests)
 set_tests_properties(core_contract PROPERTIES TIMEOUT 30)
endif()
